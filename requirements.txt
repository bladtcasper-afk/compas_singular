# Core runtime dependencies of compas_singular.
#
# This file is the single source of truth: pyproject.toml reads it via
# [tool.setuptools.dynamic], so `pip install -e .` installs exactly these.
#
# Known-good combination: compas 2.15.1 on Python 3.12, and on Rhino 8's
# CPython 3.9.10 with numpy 2.0.2 and scipy 1.13.1.
compas>=2.0,<3
# compas_singular.framefield and geometry.polyline import these directly. compas
# already requires both; they are listed so the package states its own needs.
numpy>=1.21
scipy>=1.7
# compas_singular.settings: the typed settings model, as compas_session and
# RhinoVAULT use. 2.x has wheels for Rhino 8's Python 3.9.
pydantic>=2,<3

# compas_singular.mcp adds NOTHING here on purpose. The MCP server speaks
# JSON-RPC over stdio with the standard library alone, because the official
# `mcp` SDK requires Python 3.10 and Rhino 8 ships CPython 3.9.10 -- the
# interpreter the server and the in-Rhino link both run on.
